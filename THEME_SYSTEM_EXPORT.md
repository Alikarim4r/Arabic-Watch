## 7. Rendering Architecture Notes

### 7.1 DOM layer stack (bottom → top)

```
body (page background + starfield ::before)
  └── .stage
        └── .watch (outer bezel — CSS gradients + shadows)
              ├── .ticks (#ticks) — 12 classic hour ticks, static
              ├── .ring (#ring) — rotating Arabic hour layer
              │     ├── .label × 13 (dynamic, positioned each frame)
              │     ├── .segment-arc × 72 (24 visible as ring ticks)
              │     └── .prayer-ring-marker × 6
              ├── .reader-line (#readerLine) — rotates with hour hand
              └── .clock (inner dial — CSS gradient)
                    ├── #hourMarks — .hour-dot × 12, .hour-num × 12
                    ├── #handHour, #handMinute, #handSecond
                    └── .cap
```

### 7.2 Frame rendering

- **Element:** `.watch`
- **Technique:** CSS `border-radius:50%`, `radial-gradient` background, layered `box-shadow` rings, `::before` repeating-conic-gradient texture, `::after` inner border ring
- **Not an image layer**

### 7.3 Dial rendering

- **Outer dial / bezel:** `.watch` (see above)
- **Inner dial:** `.clock` — separate 46% diameter circle, own radial-gradient and inset shadow
- **Classic 12-hour numbers:** `buildStaticMarks()` creates `.hour-num` divs with Arabic digits `12,1,2,...11`
- **Hour dots:** `.hour-dot` elements on `#hourMarks`

### 7.4 Ring rendering

- **Container:** `#ring` (`.ring`) — does NOT itself rotate via CSS transform on the container
- **Rotation model:** Each child label/marker position computed in `renderFixed()` / `renderVariable()`:
  - `ringRotation = hourAngle - arabicPointOnRing` (`exactRingRotation`)
  - Label angle: `ringRotation + idx * SLOT_ANGLE + SLOT_ANGLE/2`
  - Position: polar→cartesian via `point(angle, RADIUS)` where `RADIUS=165`
- **Ring ticks:** `.segment-arc` elements (72 created, 24 shown) at `ARC_RADIUS=181`
- **Arabic period labels:** 13 `.label` divs showing visible window ±6 periods

### 7.5 Hand rendering

- **Elements:** `#handHour`, `#handMinute`, `#handSecond` (`.hand` classes)
- **Technique:** CSS-sized rectangles with `transform-origin: 50% 100%`, rotated each frame in `render()`:
  - `secAngle = (s + ms/1000) * 6`
  - `minAngle = (m + (s+ms/1000)/60) * 6`
  - `hourAngle = hourAngleFromDate(d)` (12-hour formula × 30°)
- **Reader line:** `#readerLine` rotates with hour hand
- **Cap:** static `.cap` div, radial-gradient

### 7.6 Numbers / digital layer

- **Arabic temporal names:** `.label` on ring (text content from `timeline[].ar`)
- **Classic clock numbers:** `.hour-num` inside inner dial
- **Prayer times table:** `#prayerTimesTable` (HTML generated in `renderPrayerTimesTable`)
- **Clock readout:** `#clockReadout` — `formatClock12(d)`

### 7.7 Reflections

- **No dedicated reflection layer**
- Glow effects simulated via `box-shadow`, `text-shadow`, `filter: drop-shadow`

  function defaultRatios(count){
    return Array.from({length:count},()=>1/count);
  }

  function normalizeRatios(raw,count){
    let arr=(Array.isArray(raw) && raw.length===count ? raw.map(Number) : defaultRatios(count));
    arr=arr.map(v=>Number.isFinite(v) && v>0 ? v : 1/count);
    const minTotal=MIN_RATIO*count;
    if(minTotal>=1) return defaultRatios(count);
    arr=arr.map(v=>Math.max(MIN_RATIO,v));
    let sum=arr.reduce((a,b)=>a+b,0);
    arr=arr.map(v=>v/sum);
    for(let guard=0;guard<8;guard++){
      const low=arr.filter(v=>v<MIN_RATIO);
      if(!low.length) break;
      const fixed=arr.map(v=>Math.max(MIN_RATIO,v));
      const fixedSum=fixed.reduce((a,b)=>a+b,0);
      arr=fixed.map(v=>v/fixedSum);
    }
    const finalSum=arr.reduce((a,b)=>a+b,0) || 1;
    return arr.map(v=>v/finalSum);
  }

  function setRatio(groupId,index,value){
    const g=TEMPORAL_GROUPS.find(x=>x.id===groupId);
    if(!g) return;

    /*
      منطق التثبيت المتتابع:
      - كل ساعة قبل الساعة المعدلة تبقى كما هي.
      - الساعة الحالية تأخذ القيمة الجديدة.
      - الفرق يُوزَّع فقط على الساعات التالية.
      هذا يمنع رجوع النظام لتغيير ما ضبطه المستخدم سابقاً.
    */
    const old=normalizeRatios(customRatios[groupId],g.count);
    const beforeSum=old.slice(0,index).reduce((a,b)=>a+b,0);
    const afterCount=g.count-index-1;
    const maxAllowed=1-beforeSum-(MIN_RATIO*afterCount);
    const minAllowed=MIN_RATIO;
    const target=Math.max(minAllowed,Math.min(maxAllowed,value));
    const remaining=1-beforeSum-target;

    const next=old.slice();
    next[index]=target;

    if(afterCount>0){
      const after=old.slice(index+1);
      const afterSum=after.reduce((a,b)=>a+b,0) || afterCount;
      for(let j=index+1;j<g.count;j++){
        next[j]=(old[j]/afterSum)*remaining;
      }
    }

    customRatios[groupId]=normalizeRatios(next,g.count);
    saveRatios();
    buildTimeline();
    renderRatioControls();
  }

  function setMinutes(groupId,index,minutes){
    const g=TEMPORAL_GROUPS.find(x=>x.id===groupId);
    if(!g) return;
    const span=getGroupSpan(g).span || 1;
    setRatio(groupId,index,Number(minutes)/span);
  }

  function getGroupSpan(g){
    let from=anchors[g.from];
    let to=anchors[g.to];
    if(g.overnight && to<=from) to+=1440;
    if(g.afterMidnight){
      if(from<anchors.sunrise) from+=1440;
      if(to<from) to+=1440;
    }
    return {from,to,span:to-from};
  }

  function buildTimeline(){
    const res=[];

    for(const g of TEMPORAL_GROUPS){
      const ratios=normalizeRatios(customRatios[g.id],g.count);
      customRatios[g.id]=ratios;
      const {from,span}=getGroupSpan(g);
      let cursor=from;
      for(let i=0;i<g.count;i++){
        const dur=span*ratios[i];
        res.push({
          ar:g.names[i],
          lat:g.title,
          night:g.night,
          groupId:g.id,
          groupTitle:g.title,
          ratio:ratios[i],
          start:cursor,
          end:cursor+dur,
          duration:dur
        });
        cursor+=dur;
      }
    }

    timeline=res.map((p,i)=>({...p,idx:i}));
    PERIODS=timeline.map(p=>({ar:p.ar,lat:p.lat,night:p.night}));
    totalDuration=timeline.reduce((sum,p)=>sum+p.duration,0);

    let acc=0;
    for(const p of timeline){
      p.angleStart=(acc/totalDuration)*360;
      p.angleSize=(p.duration/totalDuration)*360;
      p.angleEnd=p.angleStart+p.angleSize;
      acc+=p.duration;
    }

    currentDateKey=todayKey();
    renderRatioControls();
  }

  function currentPeriod(minute){
    let t=minute;

    if(t<timeline[0].start) t+=1440;

    for(let i=0;i<timeline.length;i++){
      const p=timeline[i];
      let s=p.start;
      let e=p.end;
      if(e<=s)e+=1440;

      if(t>=s && t<e){
        return {
          idx:i,
          period:p,
          progress:(t-s)/(e-s),
          inside:t-s,
          absoluteMinute:t
        };
      }
    }

    return {idx:0,period:timeline[0],progress:0,inside:0,absoluteMinute:t};
  }

  function arabicPointFixed(cur){
    return cur.idx*SLOT_ANGLE + cur.progress*SLOT_ANGLE;

    const elapsedFromCycleStart=cur.absoluteMinute-timeline[0].start;
    return (elapsedFromCycleStart/totalDuration)*360;
  }

  function renderPrayerRingMarkers(ringRotation,nowMinute){
    if(!prayerRingMarkers.length) return;

    const cur=currentPeriod(nowMinute);

    for(const item of prayerRingMarkers){
      const linkedPeriod = item.data.period
        ? timeline.find(x=>x.ar===item.data.period)
        : null;

      if(linkedPeriod){
        /*
          العلامات المرتبطة بأسماء ساعات عربية:
          تظهر عند بداية الساعة المرتبطة بالصلاة، وتبقى ظاهرة طالما هذه الساعة ظاهرة على الشاشة.
        */
        const rel=signedDist(linkedPeriod.idx,cur.idx);
        const isVisible=Math.abs(rel)<=VISIBLE_HALF;

        if(isVisible){
          const startPointOnRing = linkedPeriod.idx*SLOT_ANGLE;
          const markerAngle=ringRotation+startPointOnRing;
          const pos=point(markerAngle,188);

          item.el.style.transform=
            `translate(calc(${pos.x.toFixed(2)}px - 50%), calc(${pos.y.toFixed(2)}px - 50%)) rotate(${markerAngle}deg)`;
          item.el.classList.add("show");
        }else{
          item.el.classList.remove("show");
        }

      }else{
        /*
          العشاء لا توجد له ساعة عربية بنفس الاسم.
          لذلك نحدد الفترة التي يقع داخلها وقت العشاء الحقيقي،
          ثم نظهر علامته طالما تلك الفترة ظاهرة في الشاشة.
        */
        const ishaMinute=toMin(prayerTimes.Isha);
        const ishaPeriod=currentPeriod(ishaMinute);
        const rel=signedDist(ishaPeriod.idx,cur.idx);
        const isVisible=Math.abs(rel)<=VISIBLE_HALF;

        if(isVisible){
          const markerPointOnRing=arabicPointFixed(ishaPeriod);
          const markerAngle=ringRotation+markerPointOnRing;
          const pos=point(markerAngle,188);

          item.el.style.transform=
            `translate(calc(${pos.x.toFixed(2)}px - 50%), calc(${pos.y.toFixed(2)}px - 50%)) rotate(${markerAngle}deg)`;
          item.el.classList.add("show");
        }else{
          item.el.classList.remove("show");
        }
      }
    }
  }

      const opacity=Math.pow(norm,1.35);

      label.innerHTML=labelContentForPeriod(p,currentRenderMinute);
      label.className=labelClassForPeriod(p,idx,cur,currentRenderMinute);
      label.style.transform=`translate(calc(${pos.x.toFixed(2)}px - 50%), calc(${pos.y.toFixed(2)}px - 50%))`;
      label.style.opacity=opacity.toFixed(3);
      label.style.fontSize=size.toFixed(2)+"px";
    }

    for(let i=0;i<arcTicks.length;i++){

  function renderVariable(cur,ringRotation){
    /*
      نموذج B محسّن:
      - لا نغيّر الزوايا الحقيقية للقطاعات.
      - نقلل عدد الأسماء الظاهرة لتجنب التكدس.
      - نعرض القريبة من الفترة الحالية فقط بوضوح.
      - نستخدم مستويين دائريين خفيفين بدل دفع الزوايا بشكل كبير.
    */
    const visibleOffsets=[-4,-3,-2,-1,0,1,2,3,4];

    for(let i=0;i<labels.length;i++){
      labels[i].style.opacity="0";
    }

    visibleOffsets.forEach((off,slot)=>{
      const label=labels[slot];
      const idx=mod(cur.idx+off,24);
      const p=timeline[idx];

      const rel=signedDist(idx,cur.idx);
      const absRel=Math.abs(rel);
      const norm=Math.max(0,1-absRel/5);

      const realAngle=ringRotation+p.angleStart+p.angleSize/2;

      let radius;
      if(idx===cur.idx){
        radius=RADIUS+2;
      }else if(absRel<=2){
        radius=RADIUS+18;
      }else{
        radius=RADIUS-14;
      }

      const pos=point(realAngle,radius);

      const opacity=idx===cur.idx ? 1 : Math.max(.28,Math.pow(norm,1.15));
      const size=idx===cur.idx ? 23 : Math.max(11,12+norm*6);

      label.innerHTML=labelContentForPeriod(p,currentRenderMinute);
      label.className=labelClassForPeriod(p,idx,cur,currentRenderMinute);
      label.style.transform=`translate(calc(${pos.x.toFixed(2)}px - 50%), calc(${pos.y.toFixed(2)}px - 50%))`;
      label.style.opacity=opacity.toFixed(3);
      label.style.fontSize=size.toFixed(2)+"px";
    });

    for(let i=0;i<arcTicks.length;i++){
      const el=arcTicks[i];

      if(i<24){
        const p=timeline[i];
        const angle=ringRotation+p.angleStart;
        const pos=point(angle,ARC_RADIUS);
        el.style.display="block";
        el.style.height=Math.max(8,Math.min(28,p.angleSize*.65))+"px";
        el.style.opacity=".48";
        el.style.transform=`translate(calc(${pos.x.toFixed(2)}px - 50%), calc(${pos.y.toFixed(2)}px - 50%)) rotate(${angle}deg)`;
      }else{
        el.style.display="none";
      }
    }

    modelInfo.textContent="Model B: real prayer times · cleaner labels";
    explain.innerHTML="<strong>النموذج B:</strong> القطاعات بأطوال حقيقية حسب مدة كل فترة، وتم تقليل الأسماء الظاهرة وتوزيعها على مستويين لتجنب التكدس مع بقاء تطابق عقرب الساعة.";
  }

  function render(){
    const d=simDate();

    const m=d.getMinutes();
    const s=d.getSeconds();
    const ms=d.getMilliseconds();

    const secAngle=(s+ms/1000)*6;
    const minAngle=(m+(s+ms/1000)/60)*6;
    const hourAngle=hourAngleFromDate(d);

    handSecond.style.transform=`rotate(${secAngle}deg)`;
    handMinute.style.transform=`rotate(${minAngle}deg)`;
    handHour.style.transform=`rotate(${hourAngle}deg)`;
    readerLine.style.transform=`rotate(${hourAngle}deg)`;

    clockReadout.textContent=formatClock12(d);

    const minute=minuteOfDay(d);
    currentRenderMinute=minute;
    renderPrayerTimesTable(minute);
    const cur=currentPeriod(minute);

    const pointOnArabicRing = arabicPointFixed(cur);

    const ringRotation=exactRingRotation(hourAngle,pointOnArabicRing);

    renderPrayerRingMarkers(ringRotation,minute);

    renderFixed(cur,ringRotation);

    const prevIdx=mod(cur.idx-1,24);
    prevName.textContent=timeline[prevIdx].ar;
    curName.textContent=cur.period.ar;
    curLat.textContent=`${cur.period.groupTitle} · ${cur.period.duration.toFixed(1)} دقيقة · نسبة ${(cur.period.ratio*100).toFixed(1)}%`;

    const pct=Math.round(cur.progress*100);
    progFill.style.width=`${(cur.progress*100).toFixed(2)}%`;
    progLabel.textContent=`${cur.period.ar} — ${pct}%`;

    if(todayKey()!==currentDateKey && speed===1){
      loadPrayerTimes();
    }

    requestAnimationFrame(render);
  }

  async function fetchPrayerTimes(lat,lon,label){
    try{
      const d=new Date();
      const date=`${String(d.getDate()).padStart(2,"0")}-${String(d.getMonth()+1).padStart(2,"0")}-${d.getFullYear()}`;
      const url=`https://api.aladhan.com/v1/timings/${date}?latitude=${lat}&longitude=${lon}&method=${API_METHOD}`;

      const res=await fetch(url);
      if(!res.ok) throw new Error("Prayer API failed");

      const data=await res.json();
      const t=data.data.timings;

      prayerTimes={
        Fajr:String(t.Fajr).split(" ")[0],
        Sunrise:String(t.Sunrise).split(" ")[0],
        Dhuhr:String(t.Dhuhr).split(" ")[0],
        Asr:String(t.Asr).split(" ")[0],
        Maghrib:String(t.Maghrib).split(" ")[0],
        Isha:String(t.Isha).split(" ")[0]
      };

      sourceLabel=label;
      if(data.data && data.data.meta){
        timezoneLabel=data.data.meta.timezone || timezoneLabel;
      }
      const offsetMinutes = new Date().getTimezoneOffset();
      const localHours = -offsetMinutes/60;
      timezoneOffsetText = `GMT${localHours>=0 ? "+" : ""}${localHours}`;
      locationLabel = label.includes("موقعك") ? "حسب موقعك" : "الدوحة";
      anchors=timesToAnchors(prayerTimes);
      buildTimeline();
      rebuildDynamicElements();
      ensurePrayerLine();

    }catch(e){
      prayerTimes={...FALLBACK_TIMES};
      sourceLabel="تعذر جلب المواقيت الحقيقية، تم استخدام احتياطي الدوحة";
      timezoneLabel="Asia/Qatar";
      timezoneOffsetText="GMT+3";
      locationLabel="الدوحة";
      anchors=timesToAnchors(prayerTimes);
      buildTimeline();
      rebuildDynamicElements();
      ensurePrayerLine();
    }
  }

  function loadPrayerTimes(){
    prayerTimes={...FALLBACK_TIMES};
    sourceLabel="مواقيت احتياطية للدوحة";
    timezoneLabel="Asia/Qatar";
    timezoneOffsetText="GMT+3";
    locationLabel="الدوحة";
    anchors=timesToAnchors(prayerTimes);
    buildTimeline();
    rebuildDynamicElements();
    ensurePrayerLine();

    if(navigator.geolocation){
      navigator.geolocation.getCurrentPosition(
        pos=>fetchPrayerTimes(pos.coords.latitude,pos.coords.longitude,"مواقيت حقيقية حسب موقعك"),
        ()=>fetchPrayerTimes(DOHA.lat,DOHA.lon,"مواقيت حقيقية للدوحة"),
        {enableHighAccuracy:true,timeout:9000,maximumAge:300000}
      );
    }else{
      fetchPrayerTimes(DOHA.lat,DOHA.lon,"مواقيت حقيقية للدوحة");
    }
  }
